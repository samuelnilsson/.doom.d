*** Providers

Register llama-swap as a custom eca provider, using the same host logic as
the gptel setup above: =localhost= on *DOOM_GPTEL_MAC*, or *DOOM_LLAMA_HOST*
on *DOOM_GPTEL_HOME*. On *DOOM_GPTEL_MAC* (work), also default to GitHub
Copilot's Claude Sonnet 5, overriding its reported context length to 1M
tokens (=context_length=); on *DOOM_GPTEL_HOME*, default to llama-swap's
=qwen3.5-35b= instead. The =eca= server reads its config from the
*ECA_CONFIG* environment variable, merging it with any config file, so it
can be set here without a separate JSON file.

#+begin_src emacs-lisp :tangle yes
(let* ((work-p (string-equal (getenv "DOOM_GPTEL_MAC") "t"))
       (llama-swap-host
        (cond (work-p "localhost")
              ((string-equal (getenv "DOOM_GPTEL_HOME") "t") (getenv "DOOM_LLAMA_HOST"))))
       (providers
        (append
         (when llama-swap-host
           `(("llama-swap" .
              (("api" . "openai-chat")
               ("url" . ,(concat "http://" llama-swap-host ":8080/v1"))
               ("models" . (("qwen3.5-9b" . ,(make-hash-table))
                            ("qwen3.5-35b" . ,(make-hash-table))))))))
         (when work-p
           `(("github-copilot" .
              (("models" .
                (("claude-sonnet-5" . (("context_length" . 1000000)))))))))))
       (config
        (append
         (when providers `(("providers" . ,providers)))
         (cond (work-p `(("defaultModel" . "github-copilot/claude-sonnet-5")))
               (llama-swap-host `(("defaultModel" . "llama-swap/qwen3.5-35b")))))))
  (when config
    (setenv "ECA_CONFIG" (json-encode config))))
#+end_src

** Gptel

*** Backends and models

Use gptel to interact with llama-swap language models. On Mac, use GitHub Copilot with Claude Sonnet 4.6 as the default.

#+begin_src emacs-lisp :tangle yes
(use-package! gptel
  :demand t
  :preface
  (defun gptel-for-text ()
    (if (string-equal (getenv "DOOM_GPTEL_MAC") "t")
        (progn
          (setq-local gptel-backend (gptel-get-backend "Copilot"))
          (setq-local gptel-model 'claude-sonnet-4.6)))
    (if (string-equal (getenv "DOOM_GPTEL_HOME") "t")
        (progn
          (setq-local gptel-backend (gptel-get-backend "llama-swap"))
          (setq-local gptel-model 'qwen3.5-9b)))
    (setq-local gptel-system
                "You are a crisp technical writer. Output clean Markdown with short sections, bullet lists, and examples. Prefer active voice."))
  
  (defun gptel-for-programming ()
    (if (string-equal (getenv "DOOM_GPTEL_MAC") "t")
        (progn
          (setq-local gptel-backend (gptel-get-backend "Copilot"))
          (setq-local gptel-model 'claude-sonnet-4.6)))
    (if (string-equal (getenv "DOOM_GPTEL_HOME") "t")
        (progn
          (setq-local gptel-backend (gptel-get-backend "llama-swap"))
          (setq-local gptel-model 'qwen3.5-9b)))
    (setq-local gptel-system
                "You are a careful programmer. Show minimal runnable code with comments. Note assumptions, complexity, and edge cases. Provide tests when useful."))

  :hook ((prog-mode . gptel-for-programming)
         (markdown-mode . gptel-for-text)
         (org-mode . gptel-for-text))
  :config
  (setq gptel-send-media t
        gptel-default-mode #'org-mode)
  (if (string-equal (getenv "DOOM_GPTEL_MAC") "t")
      (progn
        (gptel-make-openai "llama-swap"
          :host "localhost:8080"
          :protocol "http"
          :stream t
          :models '(qwen3.5-35b qwen3.5-9b))
        (gptel-make-gh-copilot "Copilot")
        (require 'gptel-integrations)))
  (if (string-equal (getenv "DOOM_GPTEL_HOME") "t")
      (progn
        (gptel-make-openai "llama-swap"
            :host (concat (getenv "DOOM_LLAMA_HOST") ":8080")
            :protocol "http"
            :stream t
            :models '(qwen3.5-35b qwen3.5-9b))
        (require 'gptel-integrations)))
  (if (string-equal (getenv "DOOM_GPTEL_MAC") "t")
      (setq gptel-model 'claude-sonnet-4.6
            gptel-backend (gptel-get-backend "Copilot")))
  (if (string-equal (getenv "DOOM_GPTEL_HOME") "t")
      (setq gptel-model 'qwen3.5-9b
            gptel-backend (gptel-get-backend "llama-swap"))))
#+end_src

*** Mcp servers

Configure MCP servers for gptel tool use. =microsoft-learn= provides access to official
Microsoft documentation. =pdf-reader= enables reading and extracting content from PDF files.
=atlassian-rovo= and =github= are only registered when =DOOM_GPTEL_MAC= is set, as they
require network access and credentials only available on that machine.

#+begin_src emacs-lisp :tangle yes
(use-package mcp
  :ensure t
  :after gptel
  :custom (mcp-hub-servers
           `(("microsoft-learn" . (:command "npx"
                                   :args ("-y" "mcp-remote@latest" "https://learn.microsoft.com/api/mcp")))
             ("pdf-reader" . (:command "npx"
                              :args ("@sylphx/pdf-reader-mcp")))
             ("nixos" . (:command "nix"
                              :args ("run", "github:utensils/mcp-nixos")))
             ,@(when (string-equal (getenv "DOOM_GPTEL_MAC") "t")
                 `(("atlassian-rovo" . (:command "npx"
                                        :args ("-y" "mcp-remote@latest" "https://mcp.atlassian.com/v1/mcp")))
                   ("github" . (:command "npx"
                                :args ("-y" "mcp-remote@latest" "https://api.githubcopilot.com/mcp" "--header" ,(concat "Authorization: Bearer " (string-trim (shell-command-to-string "gh auth token"))))))))))
  :config (require 'mcp-hub)
 :hook (after-init . mcp-hub-start-all-server))
#+end_src

*** Language detection

To be able to switch between writing in English and Swedish seamlessly the language is detected automatically by using the guess-language package.
It sets the correct dictionary to use for spell checking and completion depending on the language detected.

Guess language mode is only activated for text modes.
*guess-language-mode* makes sure that the correct dictionary file is used for spell checking.

#+begin_src emacs-lisp :tangle yes
(use-package! guess-language
  :hook (text-mode . guess-language-mode)
  :config
  (setq guess-language-languages '(en sv))
  (setq guess-language-min-paragraph-length 35))
  (setq guess-language-langcodes
  '((en . ("en_US" "English" nil "English"))
    (sv . ("sv_SE" "Swedish" nil "Swedish"))))
#+end_src

Hook into when a new language is detected and set the correct cape-dict file used depending on the language.

#+begin_src emacs-lisp :tangle yes
(defun set-cape-dict (lang beginning end)
  (setq cape-dict-file  (pcase lang
                          ('en (getenv "DOOM_DICT_ENG"))
                          ('sv (getenv "DOOM_DICT_SWE"))
                          )))

(add-hook 'guess-language-after-detection-functions #'set-cape-dict)
#+end_src
